let useFallbackVideo = false; // 웹캠 유무 체크 변수 async function initGame() { const vision = await FilesetResolver.forVisionTasks("https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/wasm"); poseLandmarker = await PoseLandmarker.createFromOptions(vision, { baseOptions: { modelAssetPath: "https://storage.googleapis.com/mediapipe-models/pose_landmarker/pose_landmarker_lite/float16/1/pose_landmarker_lite.task", delegate: "GPU" }, runningMode: "VIDEO", numPoses: numPlayers, minPoseDetectionConfidence: 0.5, minPosePresenceConfidence: 0.5, minTrackingConfidence: 0.5 }); // 웹캠 연결을 시도하고, 캠이 없으면 자동으로 '배경 영상 분석 모드'로 전환! try { const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user", width: { ideal: 480 }, height: { ideal: 360 } }, audio: false }); webcam.srcObject = stream; webcam.addEventListener("loadeddata", () => { statusMsg.innerText = "카메라 연결 완료! [⚡ 3초 영점 & 시작]을 누르세요."; requestAnimationFrame(renderLoop); }); } catch (err) { // PC에 웹캠이 없을 때 실행되는 구간 useFallbackVideo = true; statusMsg.innerText = "⚠️ 웹캠 없음 (PC 테스트 모드): [📂 영상(.mp4)]에 사람이 나오는 영상을 넣으면 즉시 작동합니다!"; requestAnimationFrame(renderLoop); } } let lastVideoTime = -1; function renderLoop(now) { // 웹캠이 있으면 webcam을, 없으면 bgVideo(배경 영상)를 AI에게 전달 const targetVideo = useFallbackVideo ? bgVideo : webcam; if (targetVideo.readyState >= 2 && targetVideo.currentTime !== lastVideoTime && poseLandmarker) { lastVideoTime = targetVideo.currentTime; const result = poseLandmarker.detectForVideo(targetVideo, now); ctx.clearRect(0, 0, canvas.width, canvas.height); players[0].lw = null; players[0].rw = null; players[1].lw = null; players[1].rw = null; if (result.landmarks && result.landmarks.length > 0) { // 웹캠 없을 때는 거울 반전을 끄고 영상 위치 그대로 매칭 const sortedPoses = [...result.landmarks].sort((a, b) => (1 - a[0].x) - (1 - b[0].x)); sortedPoses.forEach((lm, idx) => { if (idx < numPlayers) drawPlayer(lm, players[idx]); }); } updateAndDrawTargets(now); score1PEl.innerText = players[0].score; score2PEl.innerText = players[1].score; for (let i = particles.length - 1; i >= 0; i--) { const p = particles[i]; p.x += p.vx; p.y += p.vy; p.life -= 0.05; if (p.life <= 0) particles.splice(i, 1); else { ctx.save(); ctx.globalAlpha = p.life; ctx.fillStyle = p.color; ctx.beginPath(); ctx.arc(p.x, p.y, 5, 0, Math.PI * 2); ctx.fill(); ctx.restore(); } } } else if (useFallbackVideo && bgVideo.readyState < 2) { // 영상을 아직 안 골랐을 때도 과녁이 떨어지는지 화면 확인 가능 ctx.clearRect(0, 0, canvas.width, canvas.height); updateAndDrawTargets(now); } requestAnimationFrame(renderLoop); } initGame();
